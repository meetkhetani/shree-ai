All API requests require authentication using an API key. Include your key in the request header:

HTTP Header 
Authorization: Bearer YOUR_API_KEY
Content-Type: application/json

Chatbot API

POST /api/v1/chatbot/message

Send a message to your chatbot and receive a response

JavaScript 
const response = await fetch('https://api.shreeai.com/v1/chatbot/message', {
  method: 'POST',
  headers: {
    'Authorization': 'Bearer YOUR_API_KEY',
    'Content-Type': 'application/json'
  },
  body: JSON.stringify({
    bot_id: 'your-bot-id',
    message: 'Hello, how can you help me?',
    user_id: 'user-123',
    context: {}
  })
});

const data = await response.json();
console.log(data.reply);
Python 
import requests

url = "https://api.shreeai.com/v1/chatbot/message"
headers = {
    "Authorization": "Bearer YOUR_API_KEY",
    "Content-Type": "application/json"
}
payload = {
    "bot_id": "your-bot-id",
    "message": "Hello, how can you help me?",
    "user_id": "user-123",
    "context": {}
}

response = requests.post(url, json=payload, headers=headers)
data = response.json()
print(data['reply'])

Content Generator API

POST /api/v1/content/generate

Generate AI-powered content for various use cases

cURL 
curl -X POST https://api.shreeai.com/v1/content/generate \
  -H "Authorization: Bearer YOUR_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{
    "type": "blog_post",
    "topic": "Future of AI in Healthcare",
    "tone": "professional",
    "length": "medium",
    "keywords": ["AI", "healthcare", "innovation"]
  }'
Response 
{
  "success": true,
  "content": {
    "title": "The Future of AI in Healthcare: Transforming Patient Care",
    "body": "Artificial intelligence is revolutionizing healthcare...",
    "word_count": 850,
    "reading_time": "4 min"
  },
  "metadata": {
    "generated_at": "2025-11-08T10:30:00Z",
    "model_version": "v2.1"
  }
}

Workflow Automation API

POST /api/v1/workflow/execute

Trigger workflow automation programmatically

Node.js 
const axios = require('axios');

const executeWorkflow = async () => {
  const response = await axios.post(
    'https://api.shreeai.com/v1/workflow/execute',
    {
      workflow_id: 'wf-abc123',
      inputs: {
        email: 'user@example.com',
        action: 'send_welcome_email'
      }
    },
    {
      headers: {
        'Authorization': 'Bearer YOUR_API_KEY',
        'Content-Type': 'application/json'
      }
    }
  );
  
  return response.data;
};

executeWorkflow().then(console.log);
